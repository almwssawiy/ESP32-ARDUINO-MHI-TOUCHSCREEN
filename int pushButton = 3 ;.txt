int Led = 13 ;
int pushBttonStatus;

void setup() {
  // put your setup code here, to run once:
  pinMode(pushButton, INPUT);
  pinMode(Led,OUTPUT);
}

void loop() {   
    // put your main code here, to run repeatedly:
pushBttonStatus = digitalRead(pushButton);

if(pushBttonStatus == 1) 

{
